-- daily revenue by currency, last 14 days
SELECT date_trunc('day', placed_at) AS day,
       currency,
       sum(total) AS revenue,
       count(*) AS orders
  FROM public.orders
 WHERE status = 'paid'
   AND placed_at >= now() - interval '14 days'
 GROUP BY 1, 2
 ORDER BY 1 DESC, 3 DESC;
